name: Android

on:
  push:
    branches: [master]
  pull_request:
    branches: [master]
  workflow_dispatch:

env:
  FLUTTER_VERSION: "3.47.7"
  NDK_VERSION: "27.0.12077973"
  ANDROID_API: "34"

jobs:
  build:
    runs-on: ubuntu-latest
    timeout-minutes: 120

    steps:
      - name: Checkout (with Cactus submodule)
        uses: actions/checkout@v4
        with:
          submodules: recursive

      - name: Set up Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      # NOTE: android-actions/setup-android@v3 was removed — it fails on the
      # current runner image ("Failed to find package 'tools'"). The runner
      # already ships the Android SDK ($ANDROID_HOME) with cmdline-tools,
      # so install what we need directly with sdkmanager.
      - name: Install NDK + platform + build-tools
        run: |
          SDKMANAGER="$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager"
          yes | "$SDKMANAGER" --licenses > /dev/null
          "$SDKMANAGER" --install \
            "platform-tools" \
            "platforms;android-${{ env.ANDROID_API }}" \
            "build-tools;34.0.0" \
            "ndk;${{ env.NDK_VERSION }}"
          echo "ANDROID_NDK_HOME=$ANDROID_HOME/ndk/${{ env.NDK_VERSION }}" >> "$GITHUB_ENV"

      - name: Set up Flutter
        uses: subosito/flutter-action@v2
        with:
          flutter-version: ${{ env.FLUTTER_VERSION }}
          cache: true

      - name: Build libcactus_engine.so (arm64-v8a)
        run: |
          bash third_party/cactus/android/build.sh
          ls -la third_party/cactus/android/libcactus_engine.so

      - name: Stage native lib into jniLibs
        run: |
          mkdir -p android/app/src/main/jniLibs/arm64-v8a
          cp third_party/cactus/android/libcactus_engine.so \
             android/app/src/main/jniLibs/arm64-v8a/
          ls -la android/app/src/main/jniLibs/arm64-v8a/

      - name: flutter pub get
        run: flutter pub get

      - name: flutter analyze
        run: flutter analyze

      - name: flutter test
        run: flutter test

      - name: Build debug APK (arm64)
        run: flutter build apk --debug --target-platform android-arm64

      - name: Upload debug APK
        uses: actions/upload-artifact@v4
        with:
          name: studyreels-debug-apk
          path: build/app/outputs/flutter-apk/app-debug.apk
          if-no-files-found: error

  # On-device integration tests on an x86_64 emulator (free runner).
  # The APK is arm64-only; the google_apis x86_64 image runs it via the
  # emulator's ARM binary translation. If install fails with
  # INSTALL_FAILED_NO_MATCHING_ABIS, this job needs an x86_64 engine build.
  #
  # What this job PROVES: app launches, the model-missing gate works, the
  # native audio-extraction path yields a valid WAV, the reel-export path
  # writes a MediaStore clip — on a real Android runtime.
  #
  # What it CANNOT prove: Cactus/Qwen inference speed or quality (no model
  # bundle is installed on the emulator), real Snapdragon 7 Gen 1
  # performance, or the OS gallery-picker UI (a separate activity, not
  # drivable from flutter_test).
  integration:
    runs-on: ubuntu-latest
    # Fail fast: the whole job (emulator boot + test APK build + 3 tests)
    # fits comfortably in 25 min when healthy; longer means something hung.
    timeout-minutes: 25
    # The emulator gate is informational: it must never block the build or
    # the release. Real-device proof comes from the F55.
    continue-on-error: true

    steps:
      - name: Checkout (with Cactus submodule)
        uses: actions/checkout@v4
        with:
          submodules: recursive

      - name: Set up Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      - name: Install NDK + platform + build-tools
        run: |
          SDKMANAGER="$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager"
          yes | "$SDKMANAGER" --licenses > /dev/null
          "$SDKMANAGER" --install \
            "platform-tools" \
            "platforms;android-${{ env.ANDROID_API }}" \
            "build-tools;34.0.0" \
            "ndk;${{ env.NDK_VERSION }}"
          echo "ANDROID_NDK_HOME=$ANDROID_HOME/ndk/${{ env.NDK_VERSION }}" >> "$GITHUB_ENV"

      - name: Set up Flutter
        uses: subosito/flutter-action@v2
        with:
          flutter-version: ${{ env.FLUTTER_VERSION }}
          cache: true

      - name: Build libcactus_engine.so (arm64-v8a)
        run: |
          bash third_party/cactus/android/build.sh
          mkdir -p android/app/src/main/jniLibs/arm64-v8a
          cp third_party/cactus/android/libcactus_engine.so \
             android/app/src/main/jniLibs/arm64-v8a/

      - name: flutter pub get
        run: flutter pub get

      # Documented prerequisite for hardware-accelerated emulators on Linux
      # runners (reactivecircus/android-emulator-runner README): without
      # this, the action disables KVM and the API-34 emulator cannot boot
      # within the timeout.
      - name: Enable KVM
        run: |
          echo 'KERNEL=="kvm", GROUP="kvm", MODE="0666", OPTIONS+="static_node=kvm"' | sudo tee /etc/udev/rules.d/99-kvm4all.rules
          sudo udevadm control --reload-rules
          sudo udevadm trigger --name-match=kvm

      - name: Run integration tests on emulator
        uses: reactivecircus/android-emulator-runner@v2
        with:
          api-level: 34
          target: google_apis
          arch: x86_64
          profile: pixel_6
          # The action auto-disables KVM on some Linux runners, which makes
          # boot impossibly slow (boot timeout). Force hardware acceleration;
          # if /dev/kvm is truly absent the emulator fails fast with a clear
          # error instead of a 10-minute timeout.
          disable-linux-hw-accel: false
          # NOTE: one command per line, NO backslash continuations — the
          # action runs each line as a separate `sh -c`, so a trailing `\`
          # becomes a literal argument and breaks `flutter test`.
          # /data/local/tmp is the adb-writable scratch dir (pushing to
          # /sdcard fails with EPERM on API 34, run 37900585709); the app
          # can read it back for MediaExtractor/video_player.
          script: |
            adb push integration_test/fixtures/test_video.mp4 /data/local/tmp/test_video.mp4
            flutter test integration_test --dart-define=TEST_VIDEO_PATH=/data/local/tmp/test_video.mp4
